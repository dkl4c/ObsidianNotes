	B_{\beta}=B_{\alpha}A

$$
\begin{align}
	&A\begin{bmatrix}
	1 & 2 & 1 \\
	0 & 1 & 1 \\
	-1 & 1 & 1
	\end{bmatrix}=\begin{bmatrix}
	0 & -1 & 1 \\
	1 & 1 & 2 \\
	1 & 0 & 1
	\end{bmatrix} \\
	\implies&A\begin{bmatrix}
	1 \\
	 & 1 \\
	 &  & 1
	\end{bmatrix}=\begin{bmatrix}
	-1 & 1 & 1 \\
	2 & -3 & -2 \\
	-1 & 4 & 4
	\end{bmatrix} \\
	\implies&A=\begin{bmatrix}
	-1 & 1 & 1 \\
	2 & -3 & -2 \\
	-1 & 4 & 4
	\end{bmatrix} \\
\end{align}
